class Solution {
public:

    int findHeight(vector<vector<int>> edges) {
        queue<int> q;
        q.push(0);
        int height = 0;

        while(!q.empty()) {
            int sz = q.size();
            height++;

            for(int i = 0; i < sz; i++) {
                int node = q.front();
                q.pop();

                for(int k = 0; k < edges[node].size(); k++) {
                    q.push(edges[node][k]);
                }
            }
        }

        return height;
    }
    
    long long weightedSum(vector<int>& parent, vector<int>& nums) {
        int n = parent.size();
        vector<vector<int>> edges(n);

        for(int i = 1; i < parent.size(); i++) {
            edges[parent[i]].push_back(i);
        }
        queue<int> q;
        q.push(0);
        long long ans = 0;
        int depth = 0;
        int height = findHeight(edges);
        
        while(!q.empty()) {
            int sz = q.size();
            depth++;

            for(int i = 0; i < sz; i++) {
                int node = q.front();
                q.pop();
                ans += (height - depth + 1) * (long long)nums[node];

                for(int k = 0; k < edges[node].size(); k++) {
                    q.push(edges[node][k]);
                }
            }
        }

        return ans;
    }
};
