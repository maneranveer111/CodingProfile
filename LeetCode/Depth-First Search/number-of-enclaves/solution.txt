class Solution {
public:
    int n, m;
    int dy[4] = {1, 0, -1, 0};
    int dx[4] = {0, 1, 0, -1};

    void dfs(int i, int j, vector<vector<int>>& grid, vector<vector<int>>& vis){
        vis[i][j] = 1;

        for(int k = 0; k < 4; k++){
            int ni = i + dy[k];
            int nj = j + dx[k];

            if(ni >= 0 && ni < n && nj >= 0 && nj < m && vis[ni][nj] == 0 && grid[ni][nj] == 1)
            dfs(ni, nj, grid, vis);
        }

    }

    int numEnclaves(vector<vector<int>>& grid) {
        n = grid.size();
        m = grid[0].size();
        vector<vector<int>> vis(n, vector<int>(m, 0));
        int cnt = 0;

        for(int i = 0; i < n; i++){
            for(int j = 0; j < m; j++){
                if((i == 0 || j == 0 || i == n - 1 || j == m - 1) &&
                    vis[i][j] == 0 && grid[i][j] == 1){
                    dfs(i, j, grid, vis);
                }
            }
        }

        for(int i = 0; i < n; i++){
            for(int j = 0; j < m; j++){
                if(vis[i][j] != 1 && grid[i][j] == 1) cnt++;
            }
        }

        return cnt;
    }
};
