class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int n = grid.size();
        int m = grid[0].size();

        vector<vector<int>> vis(n, vector<int>(m, 0));
        queue<pair<pair<int, int>, int>> q;

        for(int i = 0; i < n; i++){
            for(int j = 0; j < m; j++){
                if(grid[i][j] == 2){
                    vis[i][j] = 1;
                    q.push({{i, j}, 0});
                }
            }
        }
        int t = 0;
        int dx[4] = {1, -1, 0, 0};
        int dy[4] = {0, 0, 1, -1};

        while(!q.empty()){
            int i = q.front().first.first;
            int j = q.front().first.second;
            int tc = q.front().second;
            q.pop();
            t = max(t, tc);

            for(int k = 0; k < 4; k++){
                int ni = i + dx[k];
                int nj = j + dy[k];

                if(ni >= 0 && nj >= 0 && ni < n && nj < m && grid[ni][nj] == 1 && 
                    vis[ni][nj] == 0){
                        vis[ni][nj] = 1;
                        q.push({{ni, nj}, tc + 1});
                    }
            }

        }

        for(int i = 0; i < n; i++){
            for(int j = 0; j < m; j++){
                if(grid[i][j] == 1 && vis[i][j] == 0)
                    return -1;
            }
        }


        return t;
    }
};
