class Solution {
public:
    double minPrice(vector<int>& prices, vector<int>& discounts) {
        sort(prices.rbegin(), prices.rend());
        sort(discounts.rbegin(), discounts.rend());

        double ans = 0;
        int pri = 0, dis = 0, n = prices.size(), m = discounts.size();
        
        while(pri < n && dis < m) {
            ans += (prices[pri]) * (double)(100 - discounts[dis]) / 100;
            pri++;
            dis++;
        }

        while(pri < n) {
            ans += (double)prices[pri++];
        }

        return ans;
    }
};
