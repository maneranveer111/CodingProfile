class Solution {
  public:
    vector<int> vis;
    stack<int> st;
    
    void dfs1(int node, vector<vector<int>>& adj) {
        vis[node] = 1;
        
        for(auto nei : adj[node]) {
            if(!vis[nei]) {
                dfs1(nei, adj);
            }
        }
        
        st.push(node);
    }
    
    void dfs2(int node, vector<vector<int>>& revAdj) {
        vis[node] = 1;
        
        for(auto nei : revAdj[node]) {
            if(!vis[nei]) {
                dfs2(nei, revAdj);
            }
        } 
    }
    
    int kosaraju(int v, vector<vector<int>> &edges) {
        // code here
        vector<vector<int>> adj(v), revAdj(v);
        
        for(auto edg : edges) {
            adj[edg[0]].push_back(edg[1]);
            revAdj[edg[1]].push_back(edg[0]);
        }
            
        vis.assign(v, 0);
        
        for(int i = 0; i < v; i++)
            if(!vis[i])
                dfs1(i, adj);
            
        vis.clear();
        vis.assign(v, 0);
        
        int cnt = 0;
                
        while(!st.empty()) {
            int node = st.top();
            st.pop();
            
            if(vis[node])
                continue;
            
            dfs2(node, revAdj);
            cnt++;
        }
        
        return cnt;
    }
};
